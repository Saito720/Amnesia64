cmake_minimum_required(VERSION 3.20)
project(HPL2LlamaRuntimeTests LANGUAGES CXX)

if(NOT WIN32)
    message(FATAL_ERROR "This integration test consumes the Windows x64 Release game objects.")
endif()

get_filename_component(HPL2_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
get_filename_component(WORKSPACE_ROOT "${HPL2_ROOT}/.." ABSOLUTE)
set(HPL2_LIBRARY_DIRECTORY "${WORKSPACE_ROOT}/x64/Release" CACHE PATH "Matching Release engine/dependency libraries")
set(GAME_OBJECT_DIRECTORY "${WORKSPACE_ROOT}/amnesia/src/game/x64/Release" CACHE PATH "Matching Release game object files")
set(AMNESIA_DATA_DIRECTORY "$ENV{ATDD_DIR}" CACHE PATH "Read-only retail game data used by the native runtime")
if(NOT EXISTS "${AMNESIA_DATA_DIRECTORY}/config/game.cfg")
    message(FATAL_ERROR "Set AMNESIA_DATA_DIRECTORY to an installed game with config/game.cfg.")
endif()

# Consume exactly the project's sources: an existing output directory may also
# contain stale objects from other branches/features.
file(READ "${WORKSPACE_ROOT}/amnesia/src/game/Lux.vcxproj" game_project)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${WORKSPACE_ROOT}/amnesia/src/game/Lux.vcxproj")
string(REGEX MATCHALL "<ClCompile Include=\"[^\"]+\"" game_sources "${game_project}")
set(game_objects)
foreach(source_entry IN LISTS game_sources)
    string(REGEX REPLACE "<ClCompile Include=\"([^\"]+)\"" "\\1" source_path "${source_entry}")
    string(REPLACE "\\" "/" source_path "${source_path}")
    get_filename_component(source_name "${source_path}" NAME_WE)
    if(NOT source_name STREQUAL "Main")
        list(APPEND game_objects "${GAME_OBJECT_DIRECTORY}/${source_name}.obj")
    endif()
endforeach()
if(NOT EXISTS "${GAME_OBJECT_DIRECTORY}/LuxEnemy_Llama.obj")
    message(FATAL_ERROR "Build the x64 Release game before configuring these tests.")
endif()
set_source_files_properties(${game_objects} PROPERTIES EXTERNAL_OBJECT TRUE)
add_executable(hpl2_llama_runtime_tests EnemyLlamaTests.cpp ${game_objects})
target_compile_features(hpl2_llama_runtime_tests PRIVATE cxx_std_17)
target_link_options(hpl2_llama_runtime_tests PRIVATE /LTCG /INCREMENTAL:NO)
set_property(TARGET hpl2_llama_runtime_tests PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
target_compile_definitions(hpl2_llama_runtime_tests PRIVATE USE_SDL2 USE_GAMEPAD GLEW_STATIC _NEWTON_USE_LIB IL_STATIC_LIB HAVE_LIBC AL_LIBTYPE_STATIC)
target_include_directories(hpl2_llama_runtime_tests PRIVATE "${HPL2_ROOT}/core/include" "${HPL2_ROOT}/dependencies/include" "${WORKSPACE_ROOT}/amnesia/src/game")
foreach(dependency HPL2 Newton AngelScript DevIL freealut GLEW jpeg ogg OpenAL32 png SDL2 theora vorbis zlib)
    target_link_libraries(hpl2_llama_runtime_tests PRIVATE "${HPL2_LIBRARY_DIRECTORY}/${dependency}.lib")
endforeach()
target_link_libraries(hpl2_llama_runtime_tests PRIVATE opengl32 dbghelp winmm setupapi Imm32 Version Avrt user32 gdi32 shell32 ole32 oleaut32 advapi32 uuid)
file(GLOB runtime_files "${HPL2_LIBRARY_DIRECTORY}/cudart64_*.dll" "${HPL2_LIBRARY_DIRECTORY}/cublas64_*.dll" "${HPL2_LIBRARY_DIRECTORY}/cublasLt64_*.dll")
if(runtime_files)
    add_custom_command(TARGET hpl2_llama_runtime_tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${runtime_files} "$<TARGET_FILE_DIR:hpl2_llama_runtime_tests>" VERBATIM)
endif()

enable_testing()
add_test(NAME hpl2_llama_native_runtime_contract COMMAND hpl2_llama_runtime_tests "${AMNESIA_DATA_DIRECTORY}" "${CMAKE_CURRENT_BINARY_DIR}/scratch")
set_tests_properties(hpl2_llama_native_runtime_contract PROPERTIES TIMEOUT 90)
