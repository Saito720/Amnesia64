# Use the compiler built from exactly the same sources as the runtime.
function(protobuf_generate_cpp out_sources out_headers)
    set(sources)
    set(headers)
    foreach(proto ${ARGN})
        get_filename_component(full "${proto}" ABSOLUTE)
        get_filename_component(dir "${full}" DIRECTORY)
        get_filename_component(name "${full}" NAME_WE)
        set(source "${CMAKE_CURRENT_BINARY_DIR}/${name}.pb.cc")
        set(header "${CMAKE_CURRENT_BINARY_DIR}/${name}.pb.h")
        add_custom_command(OUTPUT "${source}" "${header}"
            COMMAND protobuf::protoc --cpp_out=${CMAKE_CURRENT_BINARY_DIR} -I ${dir} ${full}
            DEPENDS protobuf::protoc "${full}" VERBATIM)
        list(APPEND sources "${source}")
        list(APPEND headers "${header}")
    endforeach()
    set(${out_sources} ${sources} PARENT_SCOPE)
    set(${out_headers} ${headers} PARENT_SCOPE)
endfunction()
