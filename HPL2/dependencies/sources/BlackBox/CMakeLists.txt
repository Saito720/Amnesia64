cmake_minimum_required(VERSION 3.15)
project(BlackBox LANGUAGES CXX RC)

add_library(BlackBox STATIC BlackBox.cpp BlackBoxUI.cpp BlackBoxReport.cpp)
target_compile_features(BlackBox PRIVATE cxx_std_14)
target_compile_definitions(BlackBox PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(BlackBox PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(BlackBox PUBLIC dbghelp comdlg32 shell32 user32 gdi32)

option(BLACKBOX_BUILD_TESTS "Build the standalone BlackBox crash tests" OFF)
if(BLACKBOX_BUILD_TESTS)
    enable_testing()
    add_executable(BlackBoxTests tests/BlackBoxTests.cpp BlackBox.rc)
    target_link_libraries(BlackBoxTests PRIVATE BlackBox)
    if(MSVC)
        target_compile_options(BlackBoxTests PRIVATE /Zi)
        target_link_options(BlackBoxTests PRIVATE /DEBUG)
    endif()
    add_test(NAME BlackBoxTests COMMAND BlackBoxTests)
    set_tests_properties(BlackBoxTests PROPERTIES TIMEOUT 90)
endif()
